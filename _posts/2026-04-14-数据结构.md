## ST表 (Sparse Table)

```cpp
class SparseTable {
    vector<vector<int>> st;
    vector<int> lg;
public:
    SparseTable(const vector<int>& a) {
        int n = a.size();
        lg.assign(n + 1, 0);
        for (int i = 2; i <= n; i++) lg[i] = lg[i >> 1] + 1;
        st.assign(lg[n] + 1, vector<int>(n));
        st[0] = a;
        for (int k = 1; k <= lg[n]; k++)
            for (int i = 0; i + (1 << k) <= n; i++)
                st[k][i] = max(st[k - 1][i], st[k - 1][i + (1 << (k - 1))]);
    }
    int query(int l, int r) {
        int k = lg[r - l + 1];
        return max(st[k][l], st[k][r - (1 << k) + 1]);
    }
};
```

    }
};

## 线段树

### 区间和 + 区间最大值

```cpp
struct SegTree {
    struct Node { int l, r, sum, mx; };
    vector<Node> tr;
    int n;
    SegTree(int n = 0) { init(n); }

    void init(int n_) {
        n = n_;
        tr.assign(4 * n + 5, {});
    }

    void build(int u, int l, int r, const vector<int>& a) {
        tr[u] = {l, r, 0, 0};
        if (l == r) { tr[u].sum = tr[u].mx = a[l]; return; }
        int mid = (l + r) >> 1;
        build(u << 1, l, mid, a);
        build(u << 1 | 1, mid + 1, r, a);
        pull(u);
    }

    void pull(int u) {
        tr[u].sum = tr[u << 1].sum + tr[u << 1 | 1].sum;
        tr[u].mx = max(tr[u << 1].mx, tr[u << 1 | 1].mx);
    }

    void modify(int u, int idx, int val) {
        if (tr[u].l == tr[u].r) { tr[u].sum = tr[u].mx = val; return; }
        int mid = (tr[u].l + tr[u].r) >> 1;
        if (idx <= mid) modify(u << 1, idx, val);
        else modify(u << 1 | 1, idx, val);
        pull(u);
    }

    int query_sum(int u, int l, int r) {
        if (tr[u].l >= l && tr[u].r <= r) return tr[u].sum;
        int mid = (tr[u].l + tr[u].r) >> 1;
        int res = 0;
        if (l <= mid) res += query_sum(u << 1, l, r);
        if (r > mid) res += query_sum(u << 1 | 1, l, r);
        return res;
    }

    int query_max(int u, int l, int r) {
        if (tr[u].l >= l && tr[u].r <= r) return tr[u].mx;
        int mid = (tr[u].l + tr[u].r) >> 1;
        int res = INT_MIN;
        if (l <= mid) res = max(res, query_max(u << 1, l, r));
        if (r > mid) res = max(res, query_max(u << 1 | 1, l, r));
        return res;
    }
};
```

### 区间加 + 区间乘 + 区间和 (Lazy Propagation)

```cpp
struct LazySegTree {
    struct Node {
        long long sum, add, mul;
    };
    vector<Node> tr;
    long long mod;

    LazySegTree(int n, long long mod = 1e9 + 7) : mod(mod) {
        tr.assign(4 * n + 5, {0, 0, 1});
    }

    void pushup(int u) {
        tr[u].sum = (tr[u << 1].sum + tr[u << 1 | 1].sum) % mod;
    }

    void apply(Node& u, long long len, long long add, long long mul) {
        u.sum = (u.sum * mul + add * len) % mod;
        u.add = (u.add * mul + add) % mod;
        u.mul = (u.mul * mul) % mod;
    }

    void pushdown(int u, int len) {
        apply(tr[u << 1], len / 2, tr[u].add, tr[u].mul);
        apply(tr[u << 1 | 1], len / 2, tr[u].add, tr[u].mul);
        tr[u].add = 0;
        tr[u].mul = 1;
    }

    // range_add 和 range_mul 配合pushdown使用
};
```

## 树状数组 (BIT)

### 基本操作

```cpp
struct BIT {
    int n;
    vector<int> bit;
    BIT(int n = 0) { init(n); }

    void init(int n_) {
        n = n_;
        bit.assign(n + 1, 0);
    }

    void add(int idx, int val) {
        for (; idx <= n; idx += idx & -idx) bit[idx] += val;
    }

    int sum(int idx) {
        int res = 0;
        for (; idx > 0; idx -= idx & -idx) res += bit[idx];
        return res;
    }

    int range_sum(int l, int r) { return sum(r) - sum(l - 1); }
};
```

### 区间加 + 区间查询

```cpp
struct BITRange {
    BIT b1, b2;
    int n;

    BITRange(int n = 0) { init(n); }

    void init(int n_) {
        n = n_;
        b1.init(n);
        b2.init(n);
    }

    void add(int idx, int val) {
        for (int i = idx; i <= n; i += i & -i) {
            b1.bit[i] += val;
            b2.bit[i] += val * (idx - 1);
        }
    }

    void range_add(int l, int r, int val) {
        add(l, val);
        add(r + 1, -val);
    }

    long long prefix_sum(int idx) {
        long long res = 0;
        for (int i = idx; i > 0; i -= i & -i) {
            res += b1.bit[i] * idx - b2.bit[i];
        }
        return res;
    }

    long long range_sum(int l, int r) {
        return prefix_sum(r) - prefix_sum(l - 1);
    }
};
```

## 单调栈

### 左边第一个更小的元素

```cpp
vector<int> nearest_smaller_left(const vector<int>& a) {
    int n = a.size();
    vector<int> res(n, -1);
    stack<int> st;
    for (int i = 0; i < n; i++) {
        while (!st.empty() && a[st.top()] >= a[i]) st.pop();
        if (!st.empty()) res[i] = st.top();
        st.push(i);
    }
    return res;
}
```

vector<int> nearest_smaller_right(const vector<int>& a) {
    int n = a.size();
    vector<int> res(n, -1);
    stack<int> st;
    for (int i = n - 1; i >= 0; i--) {
        while (!st.empty() && a[st.top()] >= a[i]) st.pop();
        if (!st.empty()) res[i] = st.top();
        st.push(i);
    }
    return res;
}

vector<int> sliding_window_min(const vector<int>& a, int k) {
    int n = a.size();
    vector<int> res;
    deque<int> dq; // 存下标
    for (int i = 0; i < n; i++) {
        while (!dq.empty() && a[dq.back()] >= a[i]) dq.pop_back();
        dq.push_back(i);
        if (i >= k - 1) {
            while (!dq.empty() && dq.front() <= i - k) dq.pop_front();
            res.push_back(a[dq.front()]);
        }
    }
    return res;
}

## 笛卡尔树

```cpp
struct CartesianTree {
    vector<int> parent, left, right;
    vector<int> inorder;

    CartesianTree(const vector<int>& a) {
        int n = a.size();
        parent.assign(n, -1);
        left.assign(n, -1);
        right.assign(n, -1);
        vector<int> st;
        for (int i = 0; i < n; i++) {
            int last = -1;
            while (!st.empty() && a[st.back()] < a[i]) {
                last = st.back();
                st.pop_back();
            }
            if (!st.empty()) {
                right[st.back()] = i;
                parent[i] = st.back();
            }
            if (last != -1) {
                left[i] = last;
                parent[last] = i;
            }
            st.push_back(i);
        }
        // 求中序遍历
        inorder_traversal(0, a);
    }

    void inorder_traversal(int u, const vector<int>& a) {
        if (u == -1) return;
        inorder_traversal(left[u], a);
        inorder.push_back(a[u]);
        inorder_traversal(right[u], a);
    }
};
```

## 平衡树 (Treap)

```cpp
struct Treap {
    struct Node {
        int key, pri, sz;
        Node *l, *r;
        Node(int k) : key(k), pri(rand()), sz(1), l(nullptr), r(nullptr) {}
    };
    Node* root = nullptr;

    int sz(Node* p) { return p ? p->sz : 0; }
    void upd(Node* p) { if (p) p->sz = sz(p->l) + sz(p->r) + 1; }

    void split(Node* p, int key, Node*& a, Node*& b) {
        if (!p) { a = b = nullptr; return; }
        if (key <= p->key) {
            split(p->l, key, a, p->l);
            b = p;
        } else {
            split(p->r, key, p->r, b);
            a = p;
        }
        upd(p);
    }

    Node* merge(Node* a, Node* b) {
        if (!a || !b) return a ? a : b;
        if (a->pri < b->pri) {
            a->r = merge(a->r, b);
            upd(a);
            return a;
        } else {
            b->l = merge(a, b->l);
            upd(b);
            return b;
        }
    }

    void insert(int key) {
        Node *a, *b;
        split(root, key, a, b);
        root = merge(merge(a, new Node(key)), b);
    }

    void erase(int key) {
        Node *a, *b, *c;
        split(root, key, a, b);
        split(b, key + 1, b, c);
        root = merge(a, c);
    }

    int kth(Node* p, int k) {
        // 第k小 (1-indexed)
        int lsz = sz(p->l);
        if (k <= lsz) return kth(p->l, k);
        if (k == lsz + 1) return p->key;
        return kth(p->r, k - lsz - 1);
    }

    int rank(Node* p, int key) {
        // key的排名
        if (!p) return 1;
        if (key <= p->key) return rank(p->l, key);
        return sz(p->l) + 1 + rank(p->r, key);
    }
};
```

## Link-Cut Tree (LCT)

```cpp
struct LCT {
    struct Node {
        int ch[2] = {0}, p = 0;
        int val, sum;
        bool rev = false;
    };
    vector<Node> t;

    LCT(int n = 0) { init(n); }

    void init(int n) {
        t.assign(n + 1, Node());
        for (int i = 1; i <= n; i++) t[i].sum = t[i].val = 0;
    }

    bool is_root(int x) {
        int p = t[x].p;
        return p == 0 || (t[p].ch[0] != x && t[p].ch[1] != x);
    }

    void pushup(int x) {
        t[x].sum = t[x].val;
        if (t[x].ch[0]) t[x].sum ^= t[t[x].ch[0]].sum;
        if (t[x].ch[1]) t[x].sum ^= t[t[x].ch[1]].sum;
    }

    void pushdown(int x) {
        if (t[x].rev) {
            swap(t[x].ch[0], t[x].ch[1]);
            if (t[x].ch[0]) t[t[x].ch[0]].rev ^= 1;
            if (t[x].ch[1]) t[t[x].ch[1]].rev ^= 1;
            t[x].rev = false;
        }
    }

    void rotate(int x) {
        int y = t[x].p, z = t[y].p;
        int dx = (t[y].ch[1] == x), dy = (z && t[z].ch[1] == y);
        int b = t[x].ch[dx ^ 1];
        if (!is_root(y)) t[z].ch[dy] = x;
        t[x].p = z;

        t[x].ch[dx ^ 1] = y;
        t[y].p = x;

        t[y].ch[dx] = b;
        if (b) t[b].p = y;

        pushup(y);
        pushup(x);
    }

    void splay(int x) {
        static vector<int> stk;
        stk.clear();
        int y = x;
        stk.push_back(y);
        while (!is_root(y)) {
            y = t[y].p;
            stk.push_back(y);
        }
        while (!stk.empty()) {
            pushdown(stk.back());
            stk.pop_back();
        }
        while (!is_root(x)) {
            y = t[x].p;
            if (!is_root(y)) {
                if ((t[t[y].ch[0]] == x) ^ (t[t[z = y].ch[0]] == y)) rotate(x);
                else rotate(y);
            }
            rotate(x);
        }
    }

    void access(int x) {
        int last = 0;
        for (int y = x; y; y = t[y].p) {
            splay(y);
            t[y].ch[1] = last;
            pushup(y);
            last = y;
        }
        splay(x);
    }

    void make_root(int x) {
        access(x);
        t[x].rev ^= 1;
    }

    int find_root(int x) {
        access(x);
        while (true) {
            pushdown(x);
            if (t[x].ch[0]) x = t[x].ch[0];
            else break;
        }
        splay(x);
        return x;
    }

    void link(int x, int y) {
        make_root(x);
        if (find_root(y) != x) t[x].p = y;
    }

    void cut(int x, int y) {
        make_root(x);
        access(y);
        if (t[y].ch[0] == x && !t[x].ch[1]) {
            t[y].ch[0] = 0;
            t[x].p = 0;
            pushup(y);
        }
    }

    long long path_query(int x, int y) {
        make_root(x);
        access(y);
        return t[y].sum;
    }
};
```

## KD-Tree

```cpp
struct KDTree {
    struct Node {
        int p[2]; // 坐标
        int mi[2], ma[2];
        int ch[2];
        int id;
    };
    vector<Node> t;
    int n, idx, dim;
    const int K = 2;

    KDTree(int n = 0) { init(n); }

    void init(int n_) {
        n = n_;
        t.assign(n + 1, Node());
        idx = 0;
    }

    int build(int l, int r, int d) {
        if (l > r) return 0;
        int mid = (l + r) >> 1;
        dim = d;
        nth_element(t.begin() + l, t.begin() + mid, t.begin() + r + 1,
            [d](const Node& a, const Node& b) { return a.p[d] < b.p[d]; });

        int cur = ++idx;
        t[cur] = t[mid];
        t[cur].ch[0] = build(l, mid - 1, d ^ 1);
        t[cur].ch[1] = build(mid + 1, r, d ^ 1);
        pushup(cur);
        return cur;
    }

    void pushup(int u) {
        for (int i = 0; i < 2; i++) {
            t[u].mi[i] = t[u].ma[i] = t[u].p[i];
            if (t[u].ch[0]) {
                t[u].mi[i] = min(t[u].mi[i], t[t[u].ch[0]].mi[i]);
                t[u].ma[i] = max(t[u].ma[i], t[t[u].ch[0]].ma[i]);
            }
            if (t[u].ch[1]) {
                t[u].mi[i] = min(t[u].mi[i], t[t[u].ch[1]].mi[i]);
                t[u].ma[i] = max(t[u].ma[i], t[t[u].ch[1]].ma[i]);
            }
        }
    }

    long long dist(int u, const vector<int>& p) {
        long long d = 0;
        for (int i = 0; i < 2; i++) {
            d += 1LL * max(0, t[u].mi[i] - p[i]) * max(0, p[i] - t[u].ma[i]);
        }
        return d;
    }

    long long ans;
    void query(int u, const vector<int>& p) {
        if (!u) return;
        long long d = 0;
        for (int i = 0; i < 2; i++) {
            d += 1LL * (t[u].p[i] - p[i]) * (t[u].p[i] - p[i]);
        }
        if (d < ans) ans = d;

        long long dl = t[u].ch[0] ? dist(t[u].ch[0], p) : (1LL << 60);
        long long dr = t[u].ch[1] ? dist(t[u].ch[1], p) : (1LL << 60);

        if (dl < dr) {
            if (dl < ans) query(t[u].ch[0], p);
            if (dr < ans) query(t[u].ch[1], p);
        } else {
            if (dr < ans) query(t[u].ch[1], p);
            if (dl < ans) query(t[u].ch[0], p);
        }
    }

    long long nearest(const vector<int>& p) {
        ans = (1LL << 60);
        query(1, p);
        return ans;
    }
};
```

## 树链剖分

```cpp
struct HLD {
    vector<vector<int>> adj;
    vector<int> parent, depth, heavy, head, pos, sz;
    int cur = 0;
    vector<int> a; // 初始值数组
    SegTree seg; // 配合线段树使用

    HLD(int n = 0) { init(n); }

    void init(int n) {
        adj.assign(n, {});
        parent.assign(n, 0);
        depth.assign(n, 0);
        heavy.assign(n, -1);
        head.assign(n, 0);
        pos.assign(n, 0);
        sz.assign(n, 0);
        cur = 0;
    }

    int dfs(int v) {
        sz[v] = 1;
        int max_sz = 0;
        for (int u : adj[v]) if (u != parent[v]) {
            parent[u] = v;
            depth[u] = depth[v] + 1;
            int child_sz = dfs(u);
            sz[v] += child_sz;
            if (child_sz > max_sz) { max_sz = child_sz; heavy[v] = u; }
        }
        return sz[v];
    }

    void decompose(int v, int h) {
        head[v] = h;
        pos[v] = cur++;
        if (heavy[v] != -1) decompose(heavy[v], h);
        for (int u : adj[v]) if (u != parent[v] && u != heavy[v])
            decompose(u, u);
    }

    void build(int root = 0) {
        dfs(root);
        decompose(root, root);
        seg.init(pos.size());
        seg.build(1, 0, pos.size() - 1, a);
    }

    // 路径查询 [u, v]
    int path_query(int u, int v) {
        int res = 0;
        while (head[u] != head[v]) {
            if (depth[head[u]] < depth[head[v]]) swap(u, v);
            res += seg.query_sum(1, pos[head[u]], pos[u]);
            u = parent[head[u]];
        }
        if (depth[u] > depth[v]) swap(u, v);
        res += seg.query_sum(1, pos[u], pos[v]);
        return res;
    }

    // 换根
    void make_root(int v) {
        // 通过重新dfs和decompose实现
    }
};
```

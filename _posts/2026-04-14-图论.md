## 最短路

### Dijkstra

```cpp
vector<long long> dijkstra(int n, int src, const vector<vector<pair<int,int>>>& g) {
    vector<long long> dist(n, LLONG_MAX);
    vector<bool> vis(n, false);
    priority_queue<pair<long long,int>, vector<pair<long long,int>>, greater<>> pq;
    dist[src] = 0;
    pq.push({0, src});
    while (!pq.empty()) {
        auto [d, u] = pq.top(); pq.pop();
        if (vis[u]) continue;
        vis[u] = true;
        for (auto [v, w] : g[u]) {
            if (dist[v] > d + w) {
                dist[v] = d + w;
                pq.push({dist[v], v});
            }
        }
    }
    return dist;
}
```

### SPFA (队列优化的Bellman-Ford)

```cpp
vector<long long> spfa(int n, int src, const vector<vector<pair<int,int>>>& g) {
    vector<long long> dist(n, LLONG_MAX);
    vector<bool> inq(n, false);
    queue<int> q;
    dist[src] = 0;
    q.push(src);
    inq[src] = true;

    while (!q.empty()) {
        int u = q.front(); q.pop();
        inq[u] = false;
        for (auto [v, w] : g[u]) {
            if (dist[v] > dist[u] + w) {
                dist[v] = dist[u] + w;
                if (!inq[v]) { q.push(v); inq[v] = true; }
            }
        }
    }
    return dist;
}

// 检测负环
bool spfa_negative_cycle(int n, const vector<vector<pair<int,int>>>& g) {
    vector<long long> dist(n, 0);
    vector<bool> inq(n, false);
    vector<int> cnt(n, 0);
    queue<int> q;
    for (int i = 0; i < n; i++) { q.push(i); inq[i] = true; }

    while (!q.empty()) {
        int u = q.front(); q.pop();
        inq[u] = false;
        for (auto [v, w] : g[u]) {
            if (dist[v] > dist[u] + w) {
                dist[v] = dist[u] + w;
                if (!inq[v]) { q.push(v); inq[v] = true; }
                if (++cnt[v] >= n) return true; // 负环
            }
        }
    }
    return false;
}
```

### Floyd

```cpp
void floyd(vector<vector<long long>>& dist, int n) {
    for (int k = 0; k < n; k++)
        for (int i = 0; i < n; i++)
            for (int j = 0; j < n; j++)
                if (dist[i][k] + dist[k][j] < dist[i][j])
                    dist[i][j] = dist[i][k] + dist[k][j];
}

// 初始化
vector<vector<long long>> init_floyd(int n) {
    vector<vector<long long>> dist(n, vector<long long>(n, LLONG_MAX / 4));
    for (int i = 0; i < n; i++) dist[i][i] = 0;
    return dist;
}
```

## 最小生成树

### Kruskal

```cpp
struct DSU {
    vector<int> p;
    DSU(int n) { p.resize(n); iota(p.begin(), p.end(), 0); }
    int find(int x) { return p[x] == x ? x : p[x] = find(p[x]); }
    bool unite(int a, int b) {
        a = find(a); b = find(b);
        if (a == b) return false;
        p[a] = b; return true;
    }
};

struct Edge { int u, v, w; };
long long kruskal(int n, vector<Edge>& edges) {
    sort(edges.begin(), edges.end(), [](const Edge& a, const Edge& b){ return a.w < b.w; });
    DSU dsu(n);
    long long mst = 0;
    int cnt = 0;
    for (auto& e : edges) {
        if (dsu.unite(e.u, e.v)) {
            mst += e.w;
            if (++cnt == n - 1) break;
        }
    }
    return mst;
}
```

### Prim

```cpp
long long prim(int n, const vector<vector<pair<int,int>>>& g) {
    vector<long long> dist(n, LLONG_MAX);
    vector<bool> vis(n, false);
    priority_queue<pair<long long,int>, vector<pair<long long,int>>, greater<>> pq;
    dist[0] = 0;
    pq.push({0, 0});
    long long mst = 0;

    while (!pq.empty()) {
        auto [d, u] = pq.top(); pq.pop();
        if (vis[u]) continue;
        vis[u] = true;
        mst += d;
        for (auto [v, w] : g[u]) {
            if (!vis[v] && dist[v] > w) {
                dist[v] = w;
                pq.push({w, v});
            }
        }
    }
    return mst;
}
```

## 连通性

### Tarjan求割点

```cpp
vector<int> cut_vertices;
int n, m, tim = 0;
vector<vector<int>> g;
vector<int> dfn, low;
vector<bool> is_cut;

void tarjan_cut(int u, int parent) {
    dfn[u] = low[u] = ++tim;
    int child = 0;
    bool is_cv = false;

    for (int v : g[u]) {
        if (!dfn[v]) {
            child++;
            tarjan_cut(v, u);
            low[u] = min(low[u], low[v]);
            if (low[v] >= dfn[u]) is_cv = true;
        } else if (v != parent) {
            low[u] = min(low[u], dfn[v]);
        }
    }

    if (parent == -1 && child > 1) is_cv = true;
    if (is_cv) is_cut[u] = true;
}
```

### Tarjan求割边 (桥)

```cpp
vector<pair<int,int>> bridges;

void tarjan_bridge(int u, int parent) {
    dfn[u] = low[u] = ++tim;

    for (int v : g[u]) {
        if (!dfn[v]) {
            tarjan_bridge(v, u);
            low[u] = min(low[u], low[v]);
            if (low[v] > dfn[u]) {
                bridges.push_back({min(u, v), max(u, v)});
            }
        } else if (v != parent) {
            low[u] = min(low[u], dfn[v]);
        }
    }
}
```

### Tarjan强连通分量

```cpp
vector<vector<int>> scc(int n, const vector<vector<int>>& g) {
    vector<int> dfn(n, 0), low(n, 0), st;
    vector<bool> in_st(n, false);
    int tim = 0;
    vector<vector<int>> sccs;

    function<void(int)> tarjan = [&](int u) {
        dfn[u] = low[u] = ++tim;
        st.push_back(u);
        in_st[u] = true;
        for (int v : g[u]) {
            if (!dfn[v]) { tarjan(v); low[u] = min(low[u], low[v]); }
            else if (in_st[v]) low[u] = min(low[u], dfn[v]);
        }
        if (low[u] == dfn[u]) {
            vector<int> scc;
            while (true) {
                int v = st.back(); st.pop_back();
                in_st[v] = false;
                scc.push_back(v);
                if (v == u) break;
            }
            sccs.push_back(scc);
        }
    };

    for (int i = 0; i < n; i++) if (!dfn[i]) tarjan(i);
    return sccs;
}
```

### 缩点 (SCC DAG)

```cpp
vector<int> scc_id;
vector<vector<int>> scc_dag(int n, const vector<vector<int>>& g) {
    vector<vector<int>> sccs = scc(n, g);
    int scc_cnt = sccs.size();
    scc_id.assign(n, 0);
    for (int i = 0; i < scc_cnt; i++)
        for (int v : sccs[i])
            scc_id[v] = i;

    vector<vector<int>> dag(scc_cnt);
    vector<vector<int>> in_deg(scc_cnt, vector<int>(scc_cnt, 0));
    for (int u = 0; u < n; u++)
        for (int v : g[u])
            if (scc_id[u] != scc_id[v] && !in_deg[scc_id[u]][scc_id[v]])
                dag[scc_id[u]].push_back(scc_id[v]), in_deg[scc_id[u]][scc_id[v]] = 1;

    return dag;
}
```

## 2-SAT

```cpp
struct TwoSAT {
    int n;
    vector<vector<int>> g, rg;
    vector<int> assignment, comp, topo;
    int scc_cnt = 0;

    TwoSAT(int n = 0) { init(n); }

    void init(int n_) {
        n = n_;
        g.assign(2 * n, {});
        rg.assign(2 * n, {});
    }

    // x为true/false, 即 x = val
    void add(int x, bool val_x, int y, bool val_y) {
        // (x == val_x) OR (y == val_y) 等价于 (!x || !val_x) -> (y && val_y)
        g[2 * x + !val_x].push_back(2 * y + val_y);
        g[2 * y + !val_y].push_back(2 * x + val_x);
    }

    void dfs1(int v) {
        comp[v] = -1;
        for (int to : g[v]) if (comp[to] == -1) dfs1(to);
        topo.push_back(v);
    }

    void dfs2(int v, int k) {
        comp[v] = k;
        for (int to : rg[v]) if (comp[to] == -1) dfs2(to, k);
    }

    bool satisfiable() {
        topo.clear();
        comp.assign(2 * n, -1);
        for (int i = 0; i < 2 * n; i++) if (comp[i] == -1) dfs1(i);
        reverse(topo.begin(), topo.end());

        scc_cnt = 0;
        for (int v : topo) if (comp[v] == -1) dfs2(v, scc_cnt++);

        assignment.assign(n, 0);
        for (int i = 0; i < n; i++)
            if (comp[2 * i] == comp[2 * i + 1]) return false;
        for (int i = 0; i < n; i++)
            assignment[i] = comp[2 * i] < comp[2 * i + 1]; // 拓扑逆序
        return true;
    }
};
```

## 差分约束

```cpp
// 差分约束系统：xi - xj <= wk 转化为 xi <= xj + wk
// 从超级源点sp向所有点连边，跑最短路
// 如果有负环则无解

struct DifferenceConstraints {
    int n;
    vector<vector<pair<int,int>>> g; // (to, w)

    DifferenceConstraints(int n = 0) { init(n); }

    void init(int n_) {
        n = n_;
        g.assign(n + 1, {});
    }

    void add_constraint(int i, int j, int w) {
        // xi - xj <= w => xi <= xj + w
        g[j].push_back({i, w});
    }

    // 返回是否有解，dist为其中一个可行解
    bool solve(vector<long long>& dist) {
        dist.assign(n + 1, 0);
        vector<int> cnt(n + 1, 0);
        vector<bool> inq(n + 1, false);
        queue<int> q;
        for (int i = 0; i <= n; i++) { q.push(i); inq[i] = true; }

        while (!q.empty()) {
            int u = q.front(); q.pop();
            inq[u] = false;
            for (auto [v, w] : g[u]) {
                if (dist[v] > dist[u] + w) {
                    dist[v] = dist[u] + w;
                    if (!inq[v]) {
                        q.push(v);
                        inq[v] = true;
                        if (++cnt[v] > n + 1) return false; // 负环，无解
                    }
                }
            }
        }
        return true;
    }
};
```

## 二分图匹配

### Hungarian Algorithm (匈牙利算法) - 最大匹配

```cpp
int n1, n2;
vector<vector<int>> g; // g[u] = {v1, v2, ...} 左边u到右边v的边
vector<int> match_r, match_l;
vector<int> dist;

bool bfs() {
    queue<int> q;
    dist.assign(n1 + 1, -1);
    for (int u = 1; u <= n1; u++) {
        if (match_l[u] == 0) { // 未匹配的左边点
            dist[u] = 0;
            q.push(u);
        }
    }
    bool found = false;
    while (!q.empty()) {
        int u = q.front(); q.pop();
        for (int v : g[u]) {
            int u2 = match_r[v];
            if (u2 && dist[u2] == -1) {
                dist[u2] = dist[u] + 1;
                q.push(u2);
            }
            if (!u2) found = true;
        }
    }
    return found;
}

bool dfs(int u) {
    for (int v : g[u]) {
        int u2 = match_r[v];
        if (!u2 || (dist[u2] == dist[u] + 1 && dfs(u2))) {
            match_l[u] = v;
            match_r[v] = u;
            return true;
        }
    }
    dist[u] = -1;
    return false;
}

int hungarian() {
    match_l.assign(n1 + 1, 0);
    match_r.assign(n2 + 1, 0);
    int matching = 0;
    while (bfs()) {
        for (int u = 1; u <= n1; u++)
            if (match_l[u] == 0 && dfs(u))
                matching++;
    }
    return matching;
}
```

### KM Algorithm - 最大权匹配

```cpp
const long long INF = (1LL << 60);

int n;
vector<vector<long long>> w; // 权值矩阵 (n x n)
vector<long long> up, vp; // 顶标
vector<int> p, way;

bool dfs_km(int v) {
    if (v == -1) return true;
    for (int to = 0; to < n; to++) {
        if (way[to] == -1) {
            long long cur = up[v] + vp[to] - w[v][to];
            if (cur == 0) {
                way[to] = v;
                if (dfs_km(p[v] == -1 ? v : p[v])) {
                    p[v] = to;
                    return true;
                }
            } else {
                // 可以在这里优化，用slack数组
            }
        }
    }
    return false;
}

long long km() {
    p.assign(n, -1);
    up.assign(n, 0);
    vp.assign(n, 0);
    for (int v = 0; v < n; v++) {
        way.assign(n, -1);
        if (!dfs_km(v)) {
            // 顶标调整
            long long delta = INF;
            for (int to = 0; to < n; to++)
                if (way[to] == -1)
                    delta = min(delta, up[v] + vp[to] - w[v][to]);
            for (int i = 0; i < n; i++) {
                if (way[i] != -1) vp[i] += delta;
                if (p[i] == v) up[i] -= delta;
            }
            // 重新dfs
        }
    }
    long long res = 0;
    for (int i = 0; i < n; i++)
        res += w[i][p[i]];
    return res;
}
```

### Hopcroft-Karp - 最大匹配 (更优复杂度)

```cpp
struct HopcroftKarp {
    int n1, n2;
    vector<vector<int>> g;
    vector<int> dist, match;

    HopcroftKarp(int n1_, int n2_) : n1(n1_), n2(n2_) {
        g.assign(n1 + 1, {});
        match.assign(n2 + 1, 0);
        dist.resize(n1 + 1);
    }

    void add_edge(int u, int v) { g[u].push_back(v); }

    bool bfs() {
        queue<int> q;
        for (int u = 1; u <= n1; u++) {
            if (match[u] == 0) dist[u] = 0, q.push(u);
            else dist[u] = -1;
        }
        bool found = false;
        while (!q.empty()) {
            int u = q.front(); q.pop();
            for (int v : g[u]) {
                int u2 = match[v];
                if (u2 != 0 && dist[u2] == -1) {
                    dist[u2] = dist[u] + 1;
                    q.push(u2);
                }
                if (u2 == 0) found = true;
            }
        }
        return found;
    }

    bool dfs(int u) {
        for (int v : g[u]) {
            int u2 = match[v];
            if (u2 == 0 || (dist[u2] == dist[u] + 1 && dfs(u2))) {
                match[u] = v;
                match[v] = u;
                return true;
            }
        }
        dist[u] = -1;
        return false;
    }

    int max_matching() {
        int matching = 0;
        while (bfs()) {
            for (int u = 1; u <= n1; u++)
                if (match[u] == 0 && dfs(u))
                    matching++;
        }
        return matching;
    }
};
```

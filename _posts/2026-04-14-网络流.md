## Dinic最大流

```cpp
struct Dinic {
    struct Edge { int to, rev; long long cap; };
    int n;
    vector<vector<Edge>> g;
    vector<int> level, it;

    Dinic(int n = 0) { init(n); }

    void init(int n_) {
        n = n_;
        g.assign(n, {});
        level.resize(n);
        it.resize(n);
    }

    void add_edge(int from, int to, long long cap) {
        Edge a{to, (int)g[to].size(), cap};
        Edge b{from, (int)g[from].size(), 0};
        g[from].push_back(a);
        g[to].push_back(b);
    }

    bool bfs(int s, int t) {
        fill(level.begin(), level.end(), -1);
        queue<int> q;
        level[s] = 0;
        q.push(s);
        while (!q.empty()) {
            int v = q.front(); q.pop();
            for (auto& e : g[v])
                if (e.cap > 0 && level[e.to] < 0) {
                    level[e.to] = level[v] + 1;
                    q.push(e.to);
                }
        }
        return level[t] >= 0;
    }

    long long dfs(int v, int t, long long f) {
        if (v == t) return f;
        for (int& i = it[v]; i < g[v].size(); i++) {
            Edge& e = g[v][i];
            if (e.cap > 0 && level[v] < level[e.to]) {
                long long d = dfs(e.to, t, min(f, e.cap));
                if (d > 0) { e.cap -= d; g[e.to][e.rev].cap += d; return d; }
            }
        }
        return 0;
    }

    long long max_flow(int s, int t) {
        long long flow = 0, INF = (1LL << 60);
        while (bfs(s, t)) {
            fill(it.begin(), it.end(), 0);
            long long f;
            while ((f = dfs(s, t, INF)) > 0) flow += f;
        }
        return flow;
    }
};
```

## ISAP (Improved Shortest Augmenting Path)

```cpp
struct ISAP {
    struct Edge { int to, rev; long long cap; };
    int n, s, t;
    vector<vector<Edge>> g;
    vector<int> level, prog, num;

    ISAP(int n = 0) { init(n); }

    void init(int n_) {
        n = n_;
        g.assign(n, {});
    }

    void add_edge(int from, int to, long long cap) {
        Edge a{to, (int)g[to].size(), cap};
        Edge b{from, (int)g[from].size(), 0};
        g[from].push_back(a);
        g[to].push_back(b);
    }

    void bfs() {
        fill(level.begin(), level.end(), -1);
        fill(num.begin(), num.end(), 0);
        queue<int> q;
        level[t] = 0;
        q.push(t);
        while (!q.empty()) {
            int v = q.front(); q.pop();
            num[level[v]]++;
            for (auto& e : g[v])
                if (e.cap > 0 && level[e.to] < 0) {
                    level[e.to] = level[v] + 1;
                    q.push(e.to);
                }
        }
    }

    long long dfs(int v, long long f) {
        if (v == t) return f;
        for (int& i = prog[v]; i < g[v].size(); i++) {
            Edge& e = g[v][i];
            if (e.cap > 0 && level[v] == level[e.to] + 1) {
                long long d = dfs(e.to, min(f, e.cap));
                if (d > 0) { e.cap -= d; g[e.to][e.rev].cap += d; return d; }
            }
        }
        if (--num[level[v]] == 0) level[s] = n; // 断层优化
        num[++level[v]]++;
        return 0;
    }

    long long max_flow(int s_, int t_) {
        s = s_; t = t_;
        level.assign(n, -1);
        prog.assign(n, 0);
        num.assign(n, 0);
        bfs();
        long long flow = 0, INF = (1LL << 60);
        while (level[s] < n) {
            flow += dfs(s, INF);
        }
        return flow;
    }
};
```

## HLPP (Highest Label Preflow Push)

```cpp
struct HLPP {
    struct Edge { int to, rev; long long cap; };
    int n, s, t, highest;
    vector<vector<Edge>> g;
    vector<long long> ex, height;
    vector<int> gap, count;
    queue<int> q;

    HLPP(int n = 0) { init(n); }

    void init(int n_) {
        n = n_;
        g.assign(n, {});
        ex.assign(n, 0);
        height.assign(n, 0);
        gap.assign(n, 0);
        count.assign(n, 0);
    }

    void add_edge(int from, int to, long long cap) {
        Edge a{to, (int)g[to].size(), cap};
        Edge b{from, (int)g[from].size(), 0};
        g[from].push_back(a);
        g[to].push_back(b);
    }

    void bfs() {
        fill(height.begin(), height.end(), -1);
        queue<int> q;
        height[t] = 0;
        q.push(t);
        while (!q.empty()) {
            int v = q.front(); q.pop();
            for (auto& e : g[v])
                if (e.cap > 0 && height[e.to] < 0) {
                    height[e.to] = height[v] + 1;
                    q.push(e.to);
                }
        }
    }

    void gap_optimize(int h) {
        for (int v = 0; v < n; v++)
            if (height[v] >= h && height[v] < n)
                count[height[v]]--, height[v] = n, count[n]++;
    }

    void push(int v, Edge& e) {
        long long d = min(ex[v], e.cap);
        e.cap -= d;
        g[e.to][e.rev].cap += d;
        ex[v] -= d;
        ex[e.to] += d;
        if (0 < ex[e.to] && ex[e.to] <= d) q.push(e.to);
    }

    void relabel(int v) {
        int min_h = INT_MAX;
        for (auto& e : g[v])
            if (e.cap > 0) min_h = min(min_h, height[e.to]);
        int h = min_h + 1;
        count[height[v]]--;
        if (count[height[v]] == 0) gap_optimize(height[v]);
        height[v] = h;
        count[h]++;
        highest = max(highest, h);
    }

    long long max_flow(int s_, int t_) {
        s = s_; t = t_;
        bfs();
        if (height[s] < 0) return 0;
        fill(count.begin(), count.end(), 0);
        for (int v = 0; v < n; v++) count[height[v]]++;
        highest = 0;
        ex[s] = (1LL << 60);
        ex[t] = -(1LL << 60);
        for (auto& e : g[s]) push(s, e);
        while (!q.empty()) {
            int v = q.front(); q.pop();
            while (ex[v] > 0) {
                if (v == t) return ex[t];
                if (height[v] == n) relabel(v);
                for (auto& e : g[v])
                    if (e.cap > 0 && height[v] == height[e.to] + 1) {
                        push(v, e);
                        if (ex[v] == 0) break;
                    }
                if (ex[v] > 0) relabel(v);
            }
        }
        return ex[t];
    }
};
```

## 最小费用最大流 (MCMF)

### SPFA版本

```cpp
struct MCMF {
    struct Edge { int to, rev, cap; long long cost; };
    int n;
    vector<vector<Edge>> g;

    MCMF(int n = 0) { init(n); }

    void init(int n_) {
        n = n_;
        g.assign(n, {});
    }

    void add_edge(int from, int to, int cap, long long cost) {
        Edge a{to, (int)g[to].size(), cap, cost};
        Edge b{from, (int)g[from].size(), 0, -cost};
        g[from].push_back(a);
        g[to].push_back(b);
    }

    pair<int, long long> min_cost_max_flow(int s, int t) {
        int flow = 0;
        long long cost = 0;
        vector<long long> dist(n), potential(n);
        vector<int> pv(n), pe(n);

        while (true) {
            fill(dist.begin(), dist.end(), (1LL << 60));
            dist[s] = 0;
            queue<int> q;
            vector<bool> inq(n, false);
            q.push(s);
            inq[s] = true;

            // SPFA
            while (!q.empty()) {
                int v = q.front(); q.pop();
                inq[v] = false;
                for (int i = 0; i < g[v].size(); i++) {
                    Edge& e = g[v][i];
                    if (e.cap > 0 && dist[e.to] > dist[v] + e.cost + potential[v] - potential[e.to]) {
                        dist[e.to] = dist[v] + e.cost + potential[v] - potential[e.to];
                        pv[e.to] = v;
                        pe[e.to] = i;
                        if (!inq[e.to]) { q.push(e.to); inq[e.to] = true; }
                    }
                }
            }

            if (dist[t] == (1LL << 60)) break;

            for (int v = 0; v < n; v++) potential[v] += dist[v];

            int add = INT_MAX;
            for (int v = t; v != s; v = pv[v])
                add = min(add, g[pv[v]][pe[v]].cap);
            flow += add;
            cost += 1LL * add * potential[t];

            for (int v = t; v != s; v = pv[v]) {
                Edge& e = g[pv[v]][pe[v]];
                e.cap -= add;
                g[v][e.rev].cap += add;
            }
        }

        return {flow, cost};
    }
};
```

### Dijkstra版本 (Johnson势)

```cpp
pair<int, long long> min_cost_max_flow(int s, int t) {
    int flow = 0;
    long long cost = 0;
    vector<long long> dist(n), potential(n, 0);
    vector<int> pv(n), pe(n);

    while (true) {
        priority_queue<pair<long long,int>, vector<pair<long long,int>>, greater<>> pq;
        fill(dist.begin(), dist.end(), (1LL << 60));
        dist[s] = 0;
        pq.push({0, s});

        while (!pq.empty()) {
            auto [d, v] = pq.top(); pq.pop();
            if (d != dist[v]) continue;
            for (int i = 0; i < g[v].size(); i++) {
                Edge& e = g[v][i];
                if (e.cap > 0) {
                    long long ndist = dist[v] + e.cost + potential[v] - potential[e.to];
                    if (dist[e.to] > ndist) {
                        dist[e.to] = ndist;
                        pv[e.to] = v;
                        pe[e.to] = i;
                        pq.push({ndist, e.to});
                    }
                }
            }
        }

        if (dist[t] == (1LL << 60)) break;

        for (int v = 0; v < n; v++) potential[v] += dist[v];

        int add = INT_MAX;
        for (int v = t; v != s; v = pv[v])
            add = min(add, g[pv[v]][pe[v]].cap);
        flow += add;
        cost += 1LL * add * potential[t];

        for (int v = t; v != s; v = pv[v]) {
            Edge& e = g[pv[v]][pe[v]];
            e.cap -= add;
            g[v][e.rev].cap += add;
        }
    }

    return {flow, cost};
}
```

## 有上下界的网络流

### 有上下界的最大流

```cpp
struct LowerBoundMaxFlow {
    Dinic dinic;
    int n, s, t;
    vector<int> in, out;

    LowerBoundMaxFlow(int n_, int s_, int t_) : n(n_), s(s_), t(t_) {
        dinic.init(n + 2);
        in.assign(n, 0);
        out.assign(n, 0);
    }

    void add_edge(int u, int v, int low, int up) {
        dinic.add_edge(u, v, up - low);
        in[v] += low;
        out[u] += low;
    }

    long long max_flow() {
        int super_s = n;
        int super_t = n + 1;

        for (int i = 0; i < n; i++) {
            if (in[i] > 0) dinic.add_edge(super_s, i, in[i]);
            if (out[i] > 0) dinic.add_edge(i, super_t, out[i]);
        }
        dinic.add_edge(t, s, (int)1e9);

        long long flow1 = dinic.max_flow(super_s, super_t);
        long long flow2 = dinic.max_flow(s, t);

        return flow2;
    }
};
```

```cpp
// 最小点覆盖 = 最大匹配数
```

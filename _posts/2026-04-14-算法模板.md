### 排序

```cpp
// 快速排序
void quick_sort(int q[], int l, int r) {
    if (l >= r) return;
    int i = l - 1, j = r + 1, x = q[l + r >> 1];
    while (i < j) {
        do i++; while (q[i] < x);
        do j--; while (q[j] > x);
        if (i < j) swap(q[i], q[j]);
    }
    quick_sort(q, l, j);
    quick_sort(q, j + 1, r);
}

// 归并排序
void merge_sort(int q[], int l, int r) {
    if (l >= r) return;
    int mid = l + r >> 1;
    merge_sort(q, l, mid);
    merge_sort(q, mid + 1, r);
    int k = 0, i = l, j = mid + 1;
    while (i <= mid && j <= r)
        if (q[i] <= q[j]) tmp[k++] = q[i++];
        else tmp[k++] = q[j++];
    while (i <= mid) tmp[k++] = q[i++];
    while (j <= r) tmp[k++] = q[j++];
    for (i = l, j = 0; i <= r; i++, j++) q[i] = tmp[j];
```

### 并查集 (DSU)

```cpp
struct DSU {
    vector<int> p, sz;
    DSU(int n) {
        p.resize(n);
        sz.assign(n, 1);
        iota(p.begin(), p.end(), 0);
    }
    int find(int x) {
        if (p[x] == x) return x;
        return p[x] = find(p[x]);
    }
    void unite(int a, int b) {
        a = find(a);
        b = find(b);
        if (a == b) return;
        if (sz[a] < sz[b]) swap(a, b);
        p[b] = a;
        sz[a] += sz[b];
    }
    bool same(int a, int b) { return find(a) == find(b); }
};
```

```cpp
vector<int> prefix_function(const string& s) {
    int n = s.size();
    vector<int> pi(n);
    for (int i = 1; i < n; i++) {
        int j = pi[i - 1];
        while (j > 0 && s[i] != s[j]) j = pi[j - 1];
        if (s[i] == s[j]) j++;
        pi[i] = j;
    }
    return pi;
}

// 在text中查找pattern
vector<int> kmp(const string& text, const string& pattern) {
    string s = pattern + '#' + text;
    vector<int> pi = prefix_function(s);
    vector<int> res;
    for (int i = pattern.size() + 1; i < s.size(); i++)
        if (pi[i] == pattern.size())
            res.push_back(i - 2 * pattern.size());
    return res;
}
```

### Manacher

```cpp
// 求回文半径数组
vector<int> manacher(const string& s) {
    string t = "#";
    for (char c : s) { t += c; t += '#'; }
    int n = t.size();
    vector<int> r(n);
    int c = 0, r_ = 0;
    for (int i = 0; i < n; i++) {
        if (i < r_) r[i] = min(r[2 * c - i], r_ - i);
        while (i - r[i] >= 0 && i + r[i] < n && t[i - r[i]] == t[i + r[i]]) r[i]++;
        if (i + r[i] > r_) { c = i; r_ = i + r[i]; }
    }
    return r;
}
```

```cpp
int phi(int x) {
    int res = x;
    for (int i = 2; i * i <= x; i++) {
        if (x % i == 0) {
            while (x % i == 0) x /= i;

    vector<int> primes;
    vector<bool> is_composite(n + 1);
    phi[1] = 1;
    for (int i = 2; i <= n; i++) {
        if (!is_composite[i]) {
            primes.push_back(i);
            phi[i] = i - 1;

            is_composite[i * p] = true;
            if (i % p == 0) {
                phi[i * p] = phi[i] * p;
                break;
            } else {
                phi[i * p] = phi[i] * (p - 1);
            }
        }
    }
    return phi;
```

```cpp
    int n, m;
    Matrix(int n = 0, int m = 0, bool ident = false) : n(n), m(m) {
```

```cpp
class SparseTable {
    vector<vector<int>> st;
    vector<int> lg;
public:
    SparseTable(const vector<int>& a) {
        int n = a.size();
        lg.assign(n + 1, 0);
        for (int i = 2; i <= n; i++) lg[i] = lg[i >> 1] + 1;
        st.assign(lg[n] + 1, vector<int>(n));
        st[0] = a;
        for (int k = 1; k <= lg[n]; k++)
            for (int i = 0; i + (1 << k) <= n; i++)
                st[k][i] = gcd(st[k - 1][i], st[k - 1][i + (1 << (k - 1))]);
    }
    int query(int l, int r) {
        int k = lg[r - l + 1];
        return gcd(st[k][l], st[k][r - (1 << k) + 1]);
    }
};
```

### 线段树

```cpp
struct SegTree {
    struct Node { int l, r, sum, mx; };
    vector<Node> tr;
    int n;
    SegTree(int n = 0) { init(n); }
    void init(int n_) { n = n_; tr.assign(4 * n + 5, {}); }
    void build(int u, int l, int r, const vector<int>& a) {
        tr[u] = {l, r, 0, 0};
        if (l == r) { tr[u].sum = tr[u].mx = a[l]; return; }
        int mid = (l + r) >> 1;
        build(u << 1, l, mid, a);
        build(u << 1 | 1, mid + 1, r, a);
        pull(u);
    }
    void pull(int u) {
        tr[u].sum = tr[u << 1].sum + tr[u << 1 | 1].sum;
        tr[u].mx = max(tr[u << 1].mx, tr[u << 1 | 1].mx);
    }
    void modify(int u, int idx, int val) {
        if (tr[u].l == tr[u].r) { tr[u].sum = tr[u].mx = val; return; }
        int mid = (tr[u].l + tr[u].r) >> 1;
        if (idx <= mid) modify(u << 1, idx, val);
        else modify(u << 1 | 1, idx, val);
        pull(u);
    }
    int query_sum(int u, int l, int r) {
        if (tr[u].l >= l && tr[u].r <= r) return tr[u].sum;
        int mid = (tr[u].l + tr[u].r) >> 1;
        int res = 0;
        if (l <= mid) res += query_sum(u << 1, l, r);
        if (r > mid) res += query_sum(u << 1 | 1, l, r);
        return res;
    }
};
```

### 平衡树 (Treap)

```cpp
struct Treap {
    struct Node {
        int key, pri, sz;
        Node *l, *r;
        Node(int k) : key(k), pri(rand()), sz(1), l(nullptr), r(nullptr) {}
    };
    Node* root = nullptr;
    int sz(Node* p) { return p ? p->sz : 0; }
    void upd(Node* p) { if (p) p->sz = sz(p->l) + sz(p->r) + 1; }
    void split(Node* p, int key, Node*& a, Node*& b) {
        if (!p) { a = b = nullptr; return; }
        if (key <= p->key) { split(p->l, key, a, p->l); b = p; }
        else { split(p->r, key, p->r, b); a = p; }
        upd(p);
    }
    Node* merge(Node* a, Node* b) {
        if (!a || !b) return a ? a : b;
        if (a->pri < b->pri) { a->r = merge(a->r, b); upd(a); return a; }
        else { b->l = merge(a, b->l); upd(b); return b; }
    }
    void insert(int key) {
        Node *a, *b;
        split(root, key, a, b);
        root = merge(merge(a, new Node(key)), b);
    }
    void erase(int key) {
        Node *a, *b, *c;
        split(root, key, a, b);
        split(b, key + 1, b, c);
        root = merge(a, c);
    }
};
```

### 树链剖分

```cpp
struct HLD {
    vector<vector<int>> adj;
    vector<int> parent, depth, heavy, head, pos, sz;
    int cur = 0;
    HLD(int n = 0) { init(n); }
    void init(int n) {
        adj.assign(n, {});
        parent.assign(n, 0);
        depth.assign(n, 0);
        heavy.assign(n, -1);
        head.assign(n, 0);
        pos.assign(n, 0);
        sz.assign(n, 0);
        cur = 0;
    }
    int dfs(int v) {
        sz[v] = 1;
        int max_sz = 0;
        for (int u : adj[v]) if (u != parent[v]) {
            parent[u] = v;
            depth[u] = depth[v] + 1;
            int child_sz = dfs(u);
            sz[v] += child_sz;
            if (child_sz > max_sz) { max_sz = child_sz; heavy[v] = u; }
        }
        return sz[v];
    }
    void decompose(int v, int h) {
        head[v] = h;
        pos[v] = cur++;
        if (heavy[v] != -1) decompose(heavy[v], h);
        for (int u : adj[v]) if (u != parent[v] && u != heavy[v])
            decompose(u, u);
    }
    // 路径查询/更新需要配合线段树
};
```

```cpp
struct ACAutomaton {
    struct Node {
        int next[26] = {0};
        int fail = 0;
        int cnt = 0;
    };
    vector<Node> t;
    ACAutomaton() { t.emplace_back(); }
    void insert(const string& s) {
        int p = 0;
        for (char c : s) {
            int idx = c - 'a';
            if (!t[p].next[idx]) { t[p].next[idx] = t.size(); t.emplace_back(); }
            p = t[p].next[idx];
        }
        t[p].cnt++;
    }
    void build() {
        queue<int> q;
        for (int i = 0; i < 26; i++)
            if (t[0].next[i]) q.push(t[0].next[i]);
        while (!q.empty()) {
            int p = q.front(); q.pop();
            for (int i = 0; i < 26; i++) {
                int u = t[p].next[i];
                if (u) {
                    t[u].fail = t[t[p].fail].next[i];
                    q.push(u);
                } else {
                    t[p].next[i] = t[t[p].fail].next[i];
                }
            }
        }
    }
    int query(const string& s) {
        int p = 0, res = 0;
        for (char c : s) {
            int idx = c - 'a';
            p = t[p].next[idx];
            for (int u = p; u && t[u].cnt != -1; u = t[u].fail)
                res += t[u].cnt, t[u].cnt = -1;
        }
        return res;
    }
};
```

### 最短路

#### Dijkstra

```cpp
vector<long long> dijkstra(int n, int src, const vector<vector<pair<int,int>>>& g) {
    vector<long long> dist(n, LLONG_MAX);
    vector<bool> vis(n, false);
    priority_queue<pair<long long,int>, vector<pair<long long,int>>, greater<>> pq;
    dist[src] = 0;
    pq.push({0, src});
    while (!pq.empty()) {
        auto [d, u] = pq.top(); pq.pop();
        if (vis[u]) continue;
        vis[u] = true;
        for (auto [v, w] : g[u]) {
            if (dist[v] > d + w) {
                dist[v] = d + w;
                pq.push({dist[v], v});
            }
        }
    }
    return dist;
}
```

```cpp
vector<long long> spfa(int n, int src, const vector<vector<pair<int,int>>>& g) {
    vector<long long> dist(n, LLONG_MAX);
    vector<bool> inq(n, false);
    queue<int> q;
    dist[src] = 0;
    q.push(src);
    inq[src] = true;
    while (!q.empty()) {
        int u = q.front(); q.pop();
        inq[u] = false;
        for (auto [v, w] : g[u]) {
            if (dist[v] > dist[u] + w) {
                dist[v] = dist[u] + w;
                if (!inq[v]) { q.push(v); inq[v] = true; }
            }
        }
    }
    return dist;
}
```

```cpp
struct DSU {
    vector<int> p;
    DSU(int n) { p.resize(n); iota(p.begin(), p.end(), 0); }
    int find(int x) { return p[x] == x ? x : p[x] = find(p[x]); }
```

### 连通性

#### Tarjan强连通分量

```cpp
vector<vector<int>> scc(int n, const vector<vector<int>>& g) {
    vector<int> dfn(n, 0), low(n, 0), st;
    vector<bool> in_st(n, false);
    int tim = 0, scc_cnt = 0;
    vector<vector<int>> sccs;

    function<void(int)> tarjan = [&](int u) {
        dfn[u] = low[u] = ++tim;
        st.push_back(u);
        in_st[u] = true;
        for (int v : g[u]) {
            if (!dfn[v]) { tarjan(v); low[u] = min(low[u], low[v]); }
            else if (in_st[v]) low[u] = min(low[u], dfn[v]);
        }
        if (low[u] == dfn[u]) {
            vector<int> scc;
            while (true) {
                int v = st.back(); st.pop_back();
                in_st[v] = false;
                scc.push_back(v);
                if (v == u) break;
            }
            sccs.push_back(scc);
        }
    };

    for (int i = 0; i < n; i++) if (!dfn[i]) tarjan(i);
    return sccs;
}
```

### 网络流

#### Dinic最大流

```cpp
struct Dinic {
    struct Edge { int to, rev; long long cap; };
    int n;
    vector<vector<Edge>> g;
    vector<int> level, it;
    Dinic(int n) : n(n), g(n), level(n), it(n) {}
    void add_edge(int from, int to, long long cap) {
        Edge a{to, (int)g[to].size(), cap};
        Edge b{from, (int)g[from].size(), 0};
        g[from].push_back(a);
        g[to].push_back(b);
    }
    bool bfs(int s, int t) {
        fill(level.begin(), level.end(), -1);
        queue<int> q;
        level[s] = 0;
        q.push(s);
        while (!q.empty()) {
            int v = q.front(); q.pop();
            for (auto& e : g[v])
                if (e.cap > 0 && level[e.to] < 0) {
                    level[e.to] = level[v] + 1;
                    q.push(e.to);
                }
        }
        return level[t] >= 0;
    }
    long long dfs(int v, int t, long long f) {
        if (v == t) return f;
        for (int& i = it[v]; i < g[v].size(); i++) {
            Edge& e = g[v][i];
            if (e.cap > 0 && level[v] < level[e.to]) {
                long long d = dfs(e.to, t, min(f, e.cap));
                if (d > 0) { e.cap -= d; g[e.to][e.rev].cap += d; return d; }
            }
        }
        return 0;
    }
    long long max_flow(int s, int t) {
        long long flow = 0, INF = LLONG_MAX;
        while (bfs(s, t)) {
            fill(it.begin(), it.end(), 0);
            long long f;
            while ((f = dfs(s, t, INF)) > 0) flow += f;
        }
        return flow;
    }
};
```

```cpp
long long dfs(int pos, int lead, int limit, int sum, vector<int>& digit) {
    if (pos == digit.size()) return sum == target;
    if (!limit && !lead && dp[pos][sum] != -1) return dp[pos][sum];
    int up = limit ? digit[pos] : 9;
    long long res = 0;
    for (int i = 0; i <= up; i++)
        res += dfs(pos + 1, lead && i == 0, limit && i == up,
                   new_sum_condition, digit);
    if (!limit && !lead) dp[pos][sum] = res;
    return res;
}
```

```cpp
struct Point {
```

```cpp
vector<Point> convex_hull(vector<Point> pts) {
    sort(pts.begin(), pts.end(), [](const Point& a, const Point& b) {
        return a.x < b.x || (a.x == b.x && a.y < b.y);
    });
    int n = pts.size(), k = 0;
    vector<Point> hull(2 * n);
    for (int i = 0; i < n; i++) {
        while (k >= 2 && (hull[k-1] - hull[k-2]).cross(pts[i] - hull[k-2]) <= 0) k--;
        hull[k++] = pts[i];
    }
    for (int i = n - 2, t = k; i >= 0; i--) {
        while (k > t && (hull[k-1] - hull[k-2]).cross(pts[i] - hull[k-2]) <= 0) k--;
        hull[k++] = pts[i];
    }
    hull.resize(k - 1);
    return hull;
}
```

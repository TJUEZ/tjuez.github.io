## 二分

### 整数二分

```cpp
// 查找第一个 >= x 的位置
int lower_bound(int x) {
    int l = 0, r = n;
    while (l < r) {
        int mid = (l + r) >> 1;
        if (a[mid] >= x) r = mid;
        else l = mid + 1;
    }
    return l;
}

// 查找第一个 > x 的位置
int upper_bound(int x) {

        else l = mid + 1;
    }
    return l;
}
```

### 实数二分

```cpp
const double EPS = 1e-7;

double binary_search_real() {
    double l = 0, r = 1e9;
    while (r - l > EPS) {
        double mid = (l + r) / 2;
        if (check(mid)) l = mid;
        else r = mid;
    }
    return l;
}
```

## 排序

### 快速排序

```cpp
void quick_sort(int q[], int l, int r) {
    if (l >= r) return;
    int i = l - 1, j = r + 1, x = q[l + r >> 1];
    while (i < j) {
        do i++; while (q[i] < x);
        do j--; while (q[j] > x);
        if (i < j) swap(q[i], q[j]);
    }
    quick_sort(q, l, j);
    quick_sort(q, j + 1, r);
}
```

### 归并排序

```cpp
int tmp[N];

void merge_sort(int q[], int l, int r) {
    if (l >= r) return;
    int mid = l + r >> 1;
    merge_sort(q, l, mid);
    merge_sort(q, mid + 1, r);
    int k = 0, i = l, j = mid + 1;
    while (i <= mid && j <= r)
        if (q[i] <= q[j]) tmp[k++] = q[i++];
        else tmp[k++] = q[j++];
    while (i <= mid) tmp[k++] = q[i++];
    while (j <= r) tmp[k++] = q[j++];
    for (i = l, j = 0; i <= r; i++, j++) q[i] = tmp[j];
}
```

### 堆排序

```cpp
void heapify(int q[], int n, int i) {
    int largest = i;
    int l = 2 * i + 1, r = 2 * i + 2;
    if (l < n && q[l] > q[largest]) largest = l;
    if (r < n && q[r] > q[largest]) largest = r;
    if (largest != i) {
        swap(q[i], q[largest]);
        heapify(q, n, largest);
    }
}

void heap_sort(int q[], int n) {
    // 构建大根堆
    for (int i = n / 2 - 1; i >= 0; i--) heapify(q, n, i);
    // 逐个取出堆顶
    for (int i = n - 1; i > 0; i--) {
        swap(q[0], q[i]);
        heapify(q, i, 0);
    }
}
```

## 并查集 (DSU)

### 基本并查集

```cpp
struct DSU {
    vector<int> p, sz;
    DSU(int n) {
        p.resize(n);
        sz.assign(n, 1);
        iota(p.begin(), p.end(), 0);
    }
    int find(int x) {
        if (p[x] == x) return x;
        return p[x] = find(p[x]);
    }
    void unite(int a, int b) {
        a = find(a);
        b = find(b);
        if (a == b) return;
        if (sz[a] < sz[b]) swap(a, b);
        p[b] = a;
        sz[a] += sz[b];
    }
    bool same(int a, int b) { return find(a) == find(b); }
    int size(int x) { return sz[find(x)]; }
};
```

### 带权并查集

```cpp
struct WeightedDSU {
    vector<int> p, d, sz;
    WeightedDSU(int n) {
        p.resize(n);
        d.assign(n, 0);
        sz.assign(n, 1);
        iota(p.begin(), p.end(), 0);
    }
    int find(int x) {
        if (p[x] == x) return x;
        int root = find(p[x]);
        d[x] += d[p[x]];
        return p[x] = root;
    }
    int weight(int x) {
        find(x);
        return d[x];
    }
    void unite(int a, int b, int w) {
        // w = weight(b) - weight(a)
        int ra = find(a), rb = find(b);
        if (ra == rb) return;
        p[rb] = ra;
        d[rb] = d[a] - d[b] + w;
        sz[ra] += sz[rb];
    }
};
```

### 并查集应用：集合个数

```cpp
// 求连通块数量
int count_components(int n, vector<vector<int>>& edges) {
    DSU dsu(n);
    for (auto& e : edges) dsu.unite(e[0], e[1]);
    int cnt = 0;
    for (int i = 0; i < n; i++) if (dsu.find(i) == i) cnt++;
    return cnt;
}
```

```cpp
// 统计1的个数
int popcount(int x) {
    int cnt = 0;
    while (x) { x &= x - 1; cnt++; }
```

## 压位DP

### 旅行商 (压位状态)

```cpp
int n;
vector<vector<int>> g; // 距离矩阵
vector<vector<int>> dp(1 << n, vector<int>(n, INF));

int solve_tsp() {
    dp[1][0] = 0; // 从城市0出发，状态只有城市0
    for (int mask = 1; mask < (1 << n); mask++) {
        for (int last = 0; last < n; last++) if (mask & (1 << last)) {
            if (dp[mask][last] == INF) continue;
            for (int nxt = 0; nxt < n; nxt++) if (!(mask & (1 << nxt))) {
                int nmask = mask | (1 << nxt);
                dp[nmask][nxt] = min(dp[nmask][nxt], dp[mask][last] + g[last][nxt]);
            }
        }
    }
    int ans = INF;
    for (int i = 1; i < n; i++)
        ans = min(ans, dp[(1 << n) - 1][i]);
    return ans;
}
```

### 斯坦纳树

```cpp
// 求连接关键点的最小生成树费用
// dp[mask][i] = 以i为根，连接了mask中关键点的最小费用
vector<vector<int>> g; // 图邻接表
int n, m; // n: 总点数, m: 关键点数量
vector<int> key; // 关键点列表
vector<vector<int>> dp(1 << m, vector<int>(n, INF));

void steiner_tree() {
    queue<int> q;
    vector<vector<bool>> inq(1 << m, vector<bool>(n, false));

    for (int mask = 1; mask < (1 << m); mask++) {
        for (int i = 0; i < n; i++) {
            // 合并子树
            for (int sub = (mask - 1) & mask; sub; sub = (sub - 1) & mask)
                dp[mask][i] = min(dp[mask][i], dp[sub][i] + dp[mask ^ sub][i]);

            if (dp[mask][i] < INF) {
                q.push(mask * n + i);
                inq[mask][i] = true;
            }
        }

        // SPFA优化
        while (!q.empty()) {
            int s = q.front() / n, v = q.front() % n;
            q.pop();
            inq[s][v] = false;
            for (int to = 0; to < n; to++) if (g[v][to] < INF) {
                int nd = dp[s][v] + g[v][to];
                if (dp[s][to] > nd) {
                    dp[s][to] = nd;
                    if (!inq[s][to]) {
                        q.push(s * n + to);
                        inq[s][to] = true;
                    }
                }
            }
        }
    }
}
```

## 启发式搜索 (A*)

```cpp
struct AStar {
    struct Node {
        int v;
        long long g, h, f;
        bool operator<(const Node& o) const { return f > o.f; }
    };

    int n, s, t;
    vector<vector<pair<int,int>>> g;
    vector<long long> dist; // 从t出发的最短路（启发函数）

    AStar(int n) : n(n), g(n) {}

    void add_edge(int u, int v, int w) {
        g[u].push_back({v, w});
    }

    void bfs() {
        // 反向BFS计算启发函数
        dist.assign(n, INF);
        queue<int> q;
        dist[t] = 0;
        q.push(t);
        while (!q.empty()) {
            int v = q.front(); q.pop();
            for (auto [u, w] : g[v]) {
                if (dist[u] > dist[v] + w) {
                    dist[u] = dist[v] + w;
                    q.push(u);
                }
            }
        }
    }

    long long astar() {
        priority_queue<Node> pq;
        vector<long long> g_val(n, INF);
        g_val[s] = 0;
        pq.push({s, 0, dist[s], dist[s]});

        while (!pq.empty()) {
            Node cur = pq.top(); pq.pop();
            if (cur.v == t) return cur.g;

            if (cur.g > g_val[cur.v]) continue;

            for (auto [to, w] : g[cur.v]) {
                long long ng = cur.g + w;
                if (ng < g_val[to]) {
                    g_val[to] = ng;
                    long long nf = ng + dist[to];
                    pq.push({to, ng, dist[to], nf});
                }
            }
        }
        return -1;
    }
};
```

## 模拟退火

```cpp
struct SimulatedAnnealing {
    double T, T_min, delta, alpha;
    int iter;
    function<double(vector<int>&)> calc; // 目标函数

    SimulatedAnnealing(double T_ = 10000, double T_min_ = 1e-8,
                       double alpha_ = 0.995, int iter_ = 100)
        : T(T_), T_min(T_min_), alpha(alpha_), iter(iter_) {}

    double solve(vector<int>& cur) {
        double cur_val = calc(cur);
        vector<int> nxt = cur;
        while (T > T_min) {
            for (int i = 0; i < iter; i++) {
                nxt = cur;
                // 随机扰动（根据具体问题设计）
                // 例如：随机交换两个位置
                int a = rand() % nxt.size();
                int b = rand() % nxt.size();
                swap(nxt[a], nxt[b]);

                double nxt_val = calc(nxt);
                delta = nxt_val - cur_val;

                if (delta < 0 || exp(-delta / T) > (double)rand() / RAND_MAX) {
                    cur = nxt;
                    cur_val = nxt_val;
                }
            }
            T *= alpha;
        }
        return cur_val;
    }
};

// 示例：求解TSP
double tsp_cost(vector<int>& path) {
    double cost = 0;
    for (int i = 0; i < path.size(); i++) {
        cost += dist[path[i]][path[(i + 1) % path.size()]];
    }
    return cost;
}
```

## 分数规划

```cpp
// 求比率最大/最小的解
// maximize: sigma(a[i]) / sigma(b[i])
// 二分答案 x，检查 sigma(a[i] - x * b[i]) >= 0

double binary_fractional_search(const vector<vector<double>>& a,
                                 const vector<vector<double>>& b,
                                 bool maximize = true) {
    double lo = 0, hi = 1e9;
    for (int it = 0; it < 60; it++) {
        double mid = (lo + hi) / 2;
        double sign = maximize ? 1 : -1;
        if (check(a, b, mid) * sign >= 0) lo = mid;
        else hi = mid;
    }
    return lo;
}

bool check(const vector<vector<double>>& a,
           const vector<vector<double>>& b, double x) {
    // 根据具体问题设计
    // 例如：选k个点最大化 a/sum - x * b/sum
    double sum = 0;
    for (...) {
        sum += a[i][j] - x * b[i][j];
    }
    return sum >= 0;
}
```

### 带修改的莫队

```cpp
struct Query {
    int l, r, t, idx, block;
    bool operator<(const Query& o) const {
        if (block != o.block) return block < o.block;
        if ((block & 1) != (o.block & 1)) return r < o.r;
        if (block & 1) return t > o.t;
        return t < o.t;
    }
};

struct Modify {
    int pos, old_val, new_val;
};

void mo_with_modify(vector<int>& a, vector<Query>& queries,
                    vector<Modify>& mods) {
    int block = pow(a.size(), 2.0 / 3);
    for (auto& q : queries) q.block = q.l / block;
    sort(queries.begin(), queries.end());

    int curL = 0, curR = -1, curT = -1;
    for (auto& q : queries) {
        while (curL > q.l) add(a[--curL]);
        while (curR < q.r) add(a[++curR]);
        while (curL < q.l) remove(a[curL++]);
        while (curR > q.r) remove(a[curR--]);
        while (curT < q.t) {
            curT++;
            if (mods[curT].pos >= curL && mods[curT].pos <= curR) {
                remove(a[mods[curT].pos]);
                a[mods[curT].pos] = mods[curT].new_val;
                add(a[mods[curT].pos]);
            } else {
                a[mods[curT].pos] = mods[curT].new_val;
            }
        }
        while (curT > q.t) {
            if (mods[curT].pos >= curL && mods[curT].pos <= curR) {
                remove(a[mods[curT].pos]);
                a[mods[curT].pos] = mods[curT].old_val;
                add(a[mods[curT].pos]);
            } else {
                a[mods[curT].pos] = mods[curT].old_val;
            }
            curT--;
        }
        ans[q.idx] = current_answer;
    }
}
```

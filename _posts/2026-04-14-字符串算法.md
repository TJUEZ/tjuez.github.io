## KMP

### 前缀函数

```cpp
vector<int> prefix_function(const string& s) {
    int n = s.size();
    vector<int> pi(n);
    for (int i = 1; i < n; i++) {
        int j = pi[i - 1];
        while (j > 0 && s[i] != s[j]) j = pi[j - 1];
        if (s[i] == s[j]) j++;
        pi[i] = j;
    }
    return pi;
}
```

### KMP匹配

```cpp
// 在text中查找pattern的所有位置
vector<int> kmp(const string& text, const string& pattern) {
    string s = pattern + '#' + text;
    vector<int> pi = prefix_function(s);
    vector<int> res;
    for (int i = pattern.size() + 1; i < s.size(); i++)
        if (pi[i] == pattern.size())
            res.push_back(i - 2 * pattern.size());
    return res;
}
```

### 扩展KMP (Z函数)

```cpp
vector<int> z_function(const string& s) {
    int n = s.size();
    vector<int> z(n);
    int l = 0, r = 0;
    for (int i = 1; i < n; i++) {
        if (i <= r) z[i] = min(r - i + 1, z[i - l]);
        while (i + z[i] < n && s[z[i]] == s[i + z[i]]) z[i]++;
        if (i + z[i] - 1 > r) {
            l = i;
            r = i + z[i] - 1;
        }
    }
    z[0] = n;
    return z;
}
```

## Manacher

```cpp
// 求回文半径数组
vector<int> manacher(const string& s) {
    string t = "#";
    for (char c : s) { t += c; t += '#'; }
    int n = t.size();
    vector<int> r(n);
    int c = 0, r_ = 0;
    for (int i = 0; i < n; i++) {
        if (i < r_) r[i] = min(r[2 * c - i], r_ - i);
        while (i - r[i] >= 0 && i + r[i] < n && t[i - r[i]] == t[i + r[i]]) r[i]++;
        if (i + r[i] > r_) { c = i; r_ = i + r[i]; }
    }
    return r;
}

// 获取所有回文子串
vector<pair<int, int>> find_palindromes(const string& s) {
    vector<int> r = manacher(s);
    vector<pair<int, int>> res;
    int n = s.size();
    for (int i = 0; i < 2 * n + 1; i++) {
        int len = r[i];
        int pos = i / 2;
        if (i % 2 == 0) {
            // 偶数长度
            int l = pos - len / 2 + 1;
            int r_ = pos + len / 2 - 1;
            if (len > 1) res.push_back({l, r_});
        } else {
            // 奇数长度
            int l = pos - len / 2;
            int r_ = pos + len / 2;
            res.push_back({l, r_});
        }
    }
    return res;
}
```

## Trie树

### 基本Trie

```cpp
struct Trie {
    struct Node {
        int next[26] = {0};
        int cnt = 0;
        int fail = 0;
    };
    vector<Node> t;
    Trie() { t.emplace_back(); }

    void insert(const string& s) {
        int p = 0;
        for (char c : s) {
            int idx = c - 'a';
            if (!t[p].next[idx]) {
                t[p].next[idx] = t.size();
                t.emplace_back();
            }
            p = t[p].next[idx];
        }
        t[p].cnt++;
    }

    bool query(const string& s) {
        int p = 0;
        for (char c : s) {
            int idx = c - 'a';
            if (!t[p].next[idx]) return false;
            p = t[p].next[idx];
        }
        return t[p].cnt > 0;
    }
};
```

### 01-Trie (异或最大值)

```cpp
struct BinaryTrie {
    struct Node {
        int ch[2] = {0};
        int cnt = 0;
    };
    vector<Node> t;
    int N; // 位数

    BinaryTrie(int N = 31) : N(N) { t.emplace_back(); }

    void insert(int x) {
        int p = 0;
        for (int i = N - 1; i >= 0; i--) {
            int bit = (x >> i) & 1;
            if (!t[p].ch[bit]) {
                t[p].ch[bit] = t.size();
                t.emplace_back();
            }
            p = t[p].ch[bit];
            t[p].cnt++;
        }
    }

    int max_xor(int x) {
        int p = 0, res = 0;
        for (int i = N - 1; i >= 0; i--) {
            int bit = (x >> i) & 1;
            int nxt = bit ^ 1;
            if (t[p].ch[nxt] && t[t[p].ch[nxt]].cnt > 0) {
                res |= (1 << i);
                p = t[p].ch[nxt];
            } else {
                p = t[p].ch[bit];
            }
        }
        return res;
    }

    void remove(int x) {
        int p = 0;
        for (int i = N - 1; i >= 0; i--) {
            int bit = (x >> i) & 1;
            p = t[p].ch[bit];
            t[p].cnt--;
        }
    }
};
```

## AC自动机

```cpp
struct ACAutomaton {
    struct Node {
        int next[26] = {0};
        int fail = 0;
        int cnt = 0;
        vector<int> output;
    };
    vector<Node> t;
    ACAutomaton() { t.emplace_back(); }

    void insert(const string& s, int id) {
        int p = 0;
        for (char c : s) {
            int idx = c - 'a';
            if (!t[p].next[idx]) {
                t[p].next[idx] = t.size();
                t.emplace_back();
            }
            p = t[p].next[idx];
        }
        t[p].cnt++;
        t[p].output.push_back(id);
    }

    void build() {
        queue<int> q;
        for (int i = 0; i < 26; i++)
            if (t[0].next[i]) q.push(t[0].next[i]);

        while (!q.empty()) {
            int p = q.front(); q.pop();
            for (int i = 0; i < 26; i++) {
                int u = t[p].next[i];
                if (u) {
                    t[u].fail = t[t[p].fail].next[i];
                    // 合并output
                    for (int v : t[t[u].fail].output)
                        t[u].output.push_back(v);
                    q.push(u);
                } else {
                    t[p].next[i] = t[t[p].fail].next[i];
                }
            }
        }
    }

    // 多模式匹配
    void query(const string& s, vector<int>& ans) {
        int p = 0;
        for (char c : s) {
            int idx = c - 'a';
            p = t[p].next[idx];
            for (int id : t[p].output)
                ans[id]++;
        }
    }
};
```

### 优化版（只记录第一次匹配）

```cpp
int query(const string& s) {
    int p = 0, res = 0;
    for (char c : s) {
        int idx = c - 'a';
        p = t[p].next[idx];
        for (int u = p; u && t[u].cnt != -1; u = t[u].fail) {
            res += t[u].cnt;
            t[u].cnt = -1; // 标记已统计
        }
    }
    return res;
}
```

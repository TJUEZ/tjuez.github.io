## 数位DP

### 框架

```cpp
long long dfs(int pos, int lead, int limit, int sum, vector<int>& digit) {
    if (pos == digit.size()) return sum == target;
    if (!limit && !lead && dp[pos][sum] != -1) return dp[pos][sum];
    int up = limit ? digit[pos] : 9;
    long long res = 0;
    for (int i = 0; i <= up; i++)
        res += dfs(pos + 1, lead && i == 0, limit && i == up,
                   new_sum_condition, digit);
    if (!limit && !lead) dp[pos][sum] = res;
    return res;
}
```

### 示例：求[1,n]中包含"13"的数的个数

```cpp
struct CountWith13 {
    long long dp[20][2][2]; // pos, has13, is_limit
    long long digit[20];

    long long solve(long long x) {
        int len = 0;
        while (x) {
            digit[++len] = x % 10;
            x /= 10;
        }
        memset(dp, -1, sizeof(dp));
        return dfs(len, 0, 1);
    }

    long long dfs(int pos, int has13, int limit) {
        if (pos == 0) return has13 == 2;
        if (!limit && dp[pos][has13][limit] != -1) return dp[pos][has13][limit];
        int up = limit ? digit[pos] : 9;
        long long res = 0;
        for (int i = 0; i <= up; i++) {
            int nh = has13;
            if (has13 == 0 && i == 1) nh = 1;
            if (has13 == 1 && i == 3) nh = 2;
            res += dfs(pos - 1, nh, limit && i == up);
        }
        if (!limit) dp[pos][has13][limit] = res;
        return res;
    }
};
```

### 示例：求[1,n]中各位数字之和等于k的数的个数

```cpp
long long dp[20][200];
long long digit[20];
long long K;

long long dfs(int pos, int sum, int limit) {
    if (sum < 0) return 0;
    if (pos == 0) return sum == 0;
    if (!limit && dp[pos][sum] != -1) return dp[pos][sum];
    int up = limit ? digit[pos] : 9;
    long long res = 0;
    for (int i = 0; i <= up; i++)
        res += dfs(pos - 1, sum - i, limit && i == up);
    if (!limit) dp[pos][sum] = res;
```

## 区间DP

### 框架

```cpp
// 区间DP常用：四边形优化 (Monotonocity)
// dp[l][r] = min(dp[l][k] + dp[k+1][r] + w[l][r])
// 满足四边形不等式时，决策单调，s[l][r-1] <= s[l][r] <= s[l+1][r]
for (int len = 2; len <= n; len++) {
    for (int l = 1; l + len - 1 <= n; l++) {
        int r = l + len - 1;
        dp[l][r] = INF;
        for (int k = s[l][r-1]; k <= s[l+1][r]; k++) {
            long long val = dp[l][k] + dp[k+1][r] + w[l][r];
            if (val < dp[l][r]) {
                dp[l][r] = val;
                s[l][r] = k;
            }
        }
    }
}
```

### 示例：合并石子

```cpp
const int MAXN = 405;
long long prefix[MAXN];
long long dp[MAXN][MAXN];
int s[MAXN][MAXN];

void stone_merge() {
    int n; // 石子堆数
    // dp[l][r] = min cost to merge [l, r] into one pile
    for (int i = 1; i <= n; i++) dp[i][i] = 0;

    for (int len = 2; len <= n; len++) {
        for (int l = 1; l + len - 1 <= n; l++) {
            int r = l + len - 1;
            dp[l][r] = LLONG_MAX;
            for (int k = l; k < r; k++) {
                long long val = dp[l][k] + dp[k+1][r] + prefix[r] - prefix[l-1];
                if (val < dp[l][r]) {
                    dp[l][r] = val;
                    s[l][r] = k;
                }
            }
        }
    }
}
```

## 树形DP

### 树的直径

```cpp
vector<vector<int>> g;
vector<int> dist;

void dfs(int u, int p, int d) {
    dist[u] = d;
    for (int v : g[u]) if (v != p)
        dfs(v, u, d + 1);
}

pair<int,int> tree_diameter() {
    dist.assign(g.size(), 0);
    dfs(0, -1, 0);
    int a = max_element(dist.begin(), dist.end()) - dist.begin();

    dist.assign(g.size(), 0);
    dfs(a, -1, 0);
    int b = max_element(dist.begin(), dist.end()) - dist.begin();

    return {a, b};
}
```

### 树形背包

```cpp
vector<vector<int>> g;
vector<vector<long long>> dp;

void dfs(int u, int p) {
    dp[u][1] = 0; // 选u

    for (int v : g[u]) if (v != p) {
        dfs(v, u);
        // 背包容量从大到小
        for (int i = sz[u]; i >= 0; i--) {
            for (int j = 0; j <= sz[v]; j++) {
                if (i + j <= sz[u])
                    dp[u][i + j] = max(dp[u][i + j], dp[u][i] + dp[v][j]);
            }
        }
        sz[u] += sz[v];
    }
}
```

### 树上最大独立集

```cpp
vector<vector<int>> g;
vector<long long> dp0, dp1; // dp0: 不选，dp1: 选

void dfs(int u, int p) {
    dp1[u] = 1;
    dp0[u] = 0;
    for (int v : g[u]) if (v != p) {
        dfs(v, u);
        dp0[u] += max(dp0[v], dp1[v]);
        dp1[u] += dp0[v];
    }
}
```

### 示例：炮兵阵地

```cpp
int n, m;
int a[105]; // 地形的二进制表示
int dp[105][1<<10][1<<10];

int can_place(int state, int h) {
    return !(state & a[h]);
}

int fit(int s1, int s2) {
    return !(s1 & s2) && !(s1 & (s2 << 1)) && !(s1 & (s2 >> 1));
}

void solve() {
    memset(dp, -1, sizeof(dp));
    dp[0][0][0] = 0;

    for (int i = 1; i <= n; i++) {
        for (int s1 = 0; s1 < (1 << m); s1++) {
            if (!can_place(s1, i)) continue;
            for (int s2 = 0; s2 < (1 << m); s2++) {
                if (dp[i-1][s2][0] < 0) continue;
                if (!can_place(s2, i-1)) continue;
                for (int s0 = 0; s0 < (1 << m); s0++) {
                    if (dp[i-2][s0][s2] < 0) continue;
                    if (!can_place(s0, i-2)) continue;
                    if (!fit(s1, s2) || !fit(s1, s0)) continue;
                    dp[i][s1][s2] = max(dp[i][s1][s2],
                        dp[i-1][s2][s0] + __builtin_popcount(s1));
                }
            }
        }
    }
}
```

### 示例：哈密顿路径

```cpp
// 推销员问题 (TSP)
// dp[mask][i] = 访问了mask中的点，最后停在i的最小花费
vector<vector<long long>> g;
int n;
vector<vector<long long>> dp(1<<n, vector<long long>(n, INF));

void hamilton() {
    dp[1][0] = 0; // 从点0出发
    for (int mask = 1; mask < (1 << n); mask++) {
        for (int i = 0; i < n; i++) if (mask & (1 << i)) {
            if (dp[mask][i] >= INF) continue;
            for (int j = 0; j < n; j++) if (!(mask & (1 << j))) {
                int nmask = mask | (1 << j);
                dp[nmask][j] = min(dp[nmask][j], dp[mask][i] + g[i][j]);
            }
        }
    }
    long long ans = INF;
    for (int i = 1; i < n; i++)
        ans = min(ans, dp[(1<<n)-1][i]);
}
```

## DP优化

### 单调队列优化

```cpp
// 转移形式: dp[i] = min/max(dp[j] + cost(j+1, i))  (i-j <= k)
// 滑动窗口最值
deque<int> dq; // 存下标
for (int i = 1; i <= n; i++) {
    while (!dq.empty() && dq.front() <= i - k) dq.pop_front();
    while (!dq.empty() && dp[dq.back()] >= dp[i]) dq.pop_back();
    dq.push_back(i);
    if (i >= k) dp[i] = dp[dq.front()] + cost[dq.front() + 1][i];
}
```

### 斜率优化

```cpp
// 转移形式: dp[i] = min(dp[j] + (long long)a[i] * b[j])
// 当b[j]单调递增时可以用单调队列
// 否则需要维护凸包

// 维护下凸包 (斜率递增)
deque<int> q;
auto slope = [&](int i, int j) {
    if (b[j] == b[i]) return (dp[i] - dp[j]) / 1.0;
    return (double)(dp[i] - dp[j]) / (b[i] - b[j]);
};

while (q.size() >= 2 && slope(q[0], q[1]) <= slope(q[1], i)) q.pop_front();

dp[i] = dp[q[0]] + (long long)a[i] * b[q[0]];
q.push_back(i);
```

### 矩阵优化

```cpp
// dp[i] = sum(dp[j] * w) => 矩阵快速幂
// 构造转移矩阵A，使 dp[i] = A * dp[i-1]
struct Matrix {
    long long a[5][5];
    Matrix(bool ident = false) {
        memset(a, 0, sizeof(a));
        if (ident) for (int i = 0; i < 5; i++) a[i][i] = 1;
    }
    Matrix operator*(const Matrix& o) const {
        Matrix res;
        for (int i = 0; i < 5; i++)
            for (int k = 0; k < 5; k++) if (a[i][k])
                for (int j = 0; j < 5; j++)
                    res.a[i][j] = (res.a[i][j] + a[i][k] * o.a[k][j]) % mod;
        return res;
    }
};

Matrix mat_pow(Matrix a, long long b) {
    Matrix res(true);
    while (b) {
        if (b & 1) res = res * a;
        a = a * a;
        b >>= 1;
    }
    return res;
}
```

## 概率DP

```cpp
// dp[i] = 期望步数
// dp[i] = sum(p[j] * dp[j]) + 1
// 写成线性方程组，高斯消元求解

vector<vector<double>> a;
vector<double> gauss(vector<vector<double>>& a, int n) {
    vector<double> x(n);
    int col = 0;
    for (int row = 0; row < n && col < n; row++, col++) {
        int sel = row;
        for (int i = row; i < n; i++)
            if (fabs(a[i][col]) > fabs(a[sel][col])) sel = i;
        if (fabs(a[sel][col]) < 1e-9) continue;
        swap(a[sel], a[row]);
        double div = a[row][col];
        for (int j = col; j <= n; j++) a[row][j] /= div;
        for (int i = 0; i < n; i++) if (i != row) {
            double factor = a[i][col];
            for (int j = col; j <= n; j++) a[i][j] -= factor * a[row][j];
        }
    }
    for (int i = 0; i < n; i++) x[i] = a[i][n];
    return x;
}
```
